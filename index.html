/** * ระบบบันทึกและรายงานผลการเช็คชื่อลูกเสือ - เนตรนารี ม.1 (เพิ่มฟิลด์ตำแหน่ง Role) * โรงเรียนสีชมพูศึกษา องค์การบริหารส่วนจังหวัดขอนแก่น */ function doPost(e) { var lock = LockService.getScriptLock(); lock.tryLock(20000); // รอคิวบันทึกสูงสุด 20 วินาที try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = getOrCreateSheet(ss); var rawData = e.postData ? e.postData.contents : null; if (!rawData && e.parameter && e.parameter.data) { rawData = e.parameter.data; } if (!rawData) { throw new Error("ไม่พบข้อมูลส่งเข้ามา (Payload is empty)"); } var data = JSON.parse(rawData); var dateStr = String(data.date || Utilities.formatDate(new Date(), "Asia/Bangkok", "yyyy-MM-dd")).trim(); var category = data.category || ""; var unit = data.unit || ""; var recordedBy = data.recordedBy ? String(data.recordedBy).trim() : "ผู้กำกับลูกเสือ"; var timestamp = Utilities.formatDate(new Date(), "Asia/Bangkok", "yyyy-MM-dd HH:mm:ss"); var lastRow = sheet.getLastRow(); var existingRowMap = {}; // ทำ Index ตรวจสอบข้อมูลซ้ำเดิม (Date + Student_ID) if (lastRow > 1) { // คอลัมน์ B (Date) และ คอลัมน์ G (Student_ID) var dateValues = sheet.getRange(2, 2, lastRow - 1, 1).getDisplayValues(); var idValues = sheet.getRange(2, 7, lastRow - 1, 1).getDisplayValues(); for (var r = 0; r < dateValues.length; r++) { var dVal = String(dateValues[r][0]).trim(); var idVal = String(idValues[r][0]).trim(); if (dVal && idVal) { existingRowMap[dVal + "_" + idVal] = r + 2; // แถวจริงในชีต } } } var rowsToAdd = []; var updatedCount = 0; var insertedCount = 0; if (data.students && data.students.length > 0) { for (var i = 0; i < data.students.length; i++) { var s = data.students[i]; var sId = String(s.id || "").trim(); var role = s.role || "สมาชิก"; var key = dateStr + "_" + sId; // ข้อมูล 10 คอลัมน์ var rowData = [ timestamp, dateStr, category, unit, s.patrol || "", role, sId, s.name || "", s.status || "มา", recordedBy ]; if (existingRowMap[key]) { // หากมีข้อมูลเดิมของวันนั้นแล้ว ให้อัปเดตแถวเดิม var targetRow = existingRowMap[key]; sheet.getRange(targetRow, 1, 1, 10).setValues([rowData]); updatedCount++; } else { // ข้อมูลใหม่ เตรียมเพิ่มต่อท้าย rowsToAdd.push(rowData); insertedCount++; } } if (rowsToAdd.length > 0) { sheet.getRange(sheet.getLastRow() + 1, 1, rowsToAdd.length, 10).setValues(rowsToAdd); } } return ContentService.createTextOutput(JSON.stringify({ "status": "success", "updated": updatedCount, "inserted": insertedCount, "message": "บันทึกเรียบร้อย (อัปเดต " + updatedCount + ", เพิ่มใหม่ " + insertedCount + ")" })).setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ "status": "error", "message": err.toString() })).setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } } // ฟังก์ชันดึงประวัติสำหรับหน้าสรุปรายงาน function doGet(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = getOrCreateSheet(ss); var lastRow = sheet.getLastRow(); if (lastRow <= 1) { return ContentService.createTextOutput(JSON.stringify({ "status": "success", "data": [] })) .setMimeType(ContentService.MimeType.JSON); } var values = sheet.getRange(2, 1, lastRow - 1, 10).getDisplayValues(); var result = []; for (var i = 0; i < values.length; i++) { result.push({ timestamp: values[i][0], date: values[i][1], category: values[i][2], unit: values[i][3], patrol: values[i][4], role: values[i][5], id: values[i][6], name: values[i][7], status: values[i][8], recordedBy: values[i][9] }); } return ContentService.createTextOutput(JSON.stringify({ "status": "success", "data": result })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ "status": "error", "message": err.toString() })) .setMimeType(ContentService.MimeType.JSON); } } // ตรวจสอบหรือสร้างชีตพร้อมหัวตาราง 10 คอลัมน์ function getOrCreateSheet(ss) { var sheet = ss.getSheetByName("Attendance_Logs"); if (!sheet) { sheet = ss.insertSheet("Attendance_Logs"); sheet.appendRow([ "Timestamp", "Date", "Category", "Unit", "Patrol", "Role", "Student_ID", "Name", "Status", "Recorded_By" ]); sheet.getRange(1, 1, 1, 10).setFontWeight("bold").setBackground("#e2e8f0"); sheet.setFrozenRows(1); } else { // ถ้ามีชีตเดิมอยู่แล้วแต่แถวหัวข้อคอลัมน์ F ยังไม่ใช่ Role ให้ตรวจสอบ/แก้ไข var colFHeader = sheet.getRange(1, 6).getValue(); if (colFHeader !== "Role") { sheet.insertColumnBefore(6); sheet.getRange(1, 6).setValue("Role").setFontWeight("bold"); } } return sheet; }
